cmake_minimum_required(VERSION 3.16)
if(ANKI_HOST)
  project(passport_anki_host C)
  set(CMAKE_C_STANDARD 11)
  enable_testing()
  add_compile_options(-Wall -Wextra -Werror)
  if(ANKI_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
  add_subdirectory(components/passport_anki)
  add_executable(passport_sim tools/passport_sim.c main/pa_protocol.c main/pa_import.c)
  target_include_directories(passport_sim PRIVATE main)
  target_link_libraries(passport_sim passport_anki)
  foreach(name core codec store)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} passport_anki)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  add_executable(test_layout tests/test_layout.c main/pa_layout.c)
  add_test(NAME layout COMMAND test_layout)
  add_executable(test_cover tests/test_cover.c main/pa_cover.c)
  target_include_directories(test_cover PRIVATE main)
  add_test(NAME cover COMMAND test_cover)
  add_executable(test_protocol tests/test_protocol.c main/pa_protocol.c)
  target_link_libraries(test_protocol passport_anki)
  add_test(NAME protocol COMMAND test_protocol)
  add_executable(test_import tests/test_import.c main/pa_import.c)
  target_link_libraries(test_import passport_anki)
  add_test(NAME import COMMAND test_import)
  add_executable(test_font_config tests/test_font_config.c)
  target_compile_definitions(test_font_config PRIVATE PA_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
  add_test(NAME font_config COMMAND test_font_config)
else()
  include($ENV{IDF_PATH}/tools/cmake/project.cmake)
  project(ai-passport-anki)
endif()
